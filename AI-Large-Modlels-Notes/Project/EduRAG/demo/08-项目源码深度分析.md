### 1.2 集成系统 (new_main.py)

**核心代码**：
```python
class IntegratedQASystem:
    def __init__(self):
        # 初始化各组件
        self.mysql_client = MySQLClient()
        self.redis_client = RedisClient()
        self.bm25_search = BM25Search(self.redis_client, self.mysql_client)
        self.client = OpenAI(api_key=..., base_url=...)  # DashScope API
        self.vector_store = VectorStore()
        self.rag_system = RAGSystem(self.vector_store, self.call_dashscope)
        
        # 初始化对话历史表
        self.init_conversation_table()
    
    def init_conversation_table(self):
        """创建conversations表存储对话历史"""
        self.mysql_client.cursor.execute("""
            CREATE TABLE IF NOT EXISTS conversations (
                id INT AUTO_INCREMENT PRIMARY KEY,
                session_id VARCHAR(36) NOT NULL,
                question TEXT NOT NULL,
                answer TEXT NOT NULL,
                timestamp DATETIME NOT NULL,
                INDEX idx_session_id (session_id)
            )
        """)
    
    def call_dashscope(self, prompt):
        """调用DashScope API生成答案（流式输出）"""
        completion = self.client.chat.completions.create(
            model=self.config.LLM_MODEL,
            messages=[
                {"role": "system", "content": "你是一个有用的助手。"},
                {"role": "user", "content": prompt},
            ],
            stream=True
        )
        for chunk in completion:
            if chunk.choices and chunk.choices[0].delta.content:
                yield chunk.choices[0].delta.content
    
    def query(self, query, source_filter=None, session_id=None):
        """查询集成系统，支持对话历史和流式输出"""
        # 获取对话历史
        history = self.get_session_history(session_id) if session_id else []
        
        # 首先尝试MySQL精确查询
        answer, need_rag = self.bm25_search.search(query, threshold=0.85)
        
        if answer:
            # MySQL找到精确答案，直接返回
            if session_id:
                self.update_session_history(session_id, query, answer)
            yield answer, True
        elif need_rag:
            # 需要使用RAG系统，支持流式输出
            collected_answer = ""
            for token in self.rag_system.generate_answer(query, source_filter=source_filter, history=history):
                collected_answer += token
                yield token, False
            
            # 完整答案收集完毕，更新会话历史
            if session_id:
                self.update_session_history(session_id, query, collected_answer)
            yield "", True
```

**关键设计**：
1. **双系统架构**：MySQL FAQ + Milvus RAG
2. **流式输出**：使用generator实现
3. **对话历史**：存储在MySQL中，保留最近5轮

### 1.3 BM25检索 (bm25_search.py)

**核心代码**：
```python
from rank_bm25 import BM25Okapi
import numpy as np

class BM25Search:
    def __init__(self, redis_client, mysql_client):
        self.redis_client = redis_client
        self.mysql_client = mysql_client
        self._load_data()
    
    def _load_data(self):
        """从Redis或MySQL加载数据"""
        # 先从Redis获取
        self.original_questions = self.redis_client.get_data("qa_original_questions")
        tokenized_questions = self.redis_client.get_data("qa_tokenized_questions")
        
        # Redis中没有，从MySQL加载
        if not self.original_questions or not tokenized_questions:
            self.original_questions = self.mysql_client.fetch_questions()
            tokenized_questions = [preprocess_text(q[0]) for q in self.original_questions]
            
            # 存储到Redis
            self.redis_client.set_data("qa_original_questions", [(q[0]) for q in self.original_questions])
            self.redis_client.set_data("qa_tokenized_questions", tokenized_questions)
        
        # 初始化BM25模型
        self.bm25 = BM25Okapi(tokenized_questions)
    
    def _softmax(self, scores):
        """计算Softmax分数，将BM25分数映射到0-1"""
        exp_scores = np.exp(scores - np.max(scores))
        return exp_scores / exp_scores.sum()
    
    def search(self, query, threshold=0.85):
        """搜索查询"""
        # 检查Redis缓存
        cached_answer = self.redis_client.get_answer(query)
        if cached_answer:
            return cached_answer, False
        
        # 分词查询
        query_tokens = preprocess_text(query)
        
        # 计算BM25分数
        scores = self.bm25.get_scores(query_tokens)
        
        # 计算Softmax分数
        softmax_scores = self._softmax(scores)
        
        # 获取最高分
        best_idx = softmax_scores.argmax()
        best_score = softmax_scores[best_idx]
        
        # 检查是否超过阈值
        if best_score >= threshold:
            original_question = self.original_questions[best_idx]
            answer = self.mysql_client.fetch_answer(original_question)
            if answer:
                # 缓存答案
                self.redis_client.set_data(f"answer:{query}", answer)
                return answer, False
        
        return None, True
```

**关键设计**：
1. **数据缓存**：FAQ数据存储在Redis中，避免重复查询MySQL
2. **Softmax归一化**：将BM25分数映射到0-1，便于设置阈值
3. **答案缓存**：查询结果缓存到Redis

### 1.4 RAG系统 (new_rag_system.py)

**核心代码**：
```python
class RAGSystem:
    def __init__(self, vector_store, llm):
        self.vector_store = vector_store
        self.llm = llm
        self.rag_prompt = RAGPrompts.rag_prompt()
        self.query_classifier = QueryClassifier()
        self.strategy_selector = StrategySelector()
    
    def _retrieve_with_hyde(self, query):
        """使用HyDE策略进行检索"""
        # 生成假设答案
        hypo_answer = self.llm(hyde_prompt.format(query=query))
        hypo_answer = "".join(hypo_answer)
        
        # 使用假设答案进行检索
        return self.vector_store.hybrid_search_with_rerank(hypo_answer, k=conf.RETRIEVAL_K)
    
    def _retrieve_with_subqueries(self, query):
        """使用子查询策略进行检索"""
        # 生成子查询
        subqueries_text = self.llm(subquery_prompt.format(query=query))
        subqueries = [q.strip() for q in subqueries_text.split("\n") if q.strip()]
        
        # 每个子查询都执行检索
        all_docs = []
        for sub_q in subqueries:
            docs = self.vector_store.hybrid_search_with_rerank(sub_q, k=conf.RETRIEVAL_K)
            all_docs.extend(docs)
        
        # 去重
        unique_docs_dict = {doc.page_content: doc for doc in all_docs}
        return list(unique_docs_dict.values())
    
    def _retrieve_with_backtracking(self, query):
        """使用回溯问题策略进行检索"""
        # 生成简化问题
        simplified_query = self.llm(backtrack_prompt.format(query=query))
        simplified_query = "".join(simplified_query)
        
        return self.vector_store.hybrid_search_with_rerank(simplified_query, k=conf.RETRIEVAL_K)
    
    def retrieve_and_merge(self, query, source_filter=None, strategy=None):
        """检索并合并相关文档"""
        if not strategy:
            strategy = self.strategy_selector.select_strategy(query)
        
        if strategy == "回溯问题检索":
            ranked_sub_chunks = self._retrieve_with_backtracking(query)
        elif strategy == "子查询检索":
            ranked_sub_chunks = self._retrieve_with_subqueries(query)
        elif strategy == "假设问题检索":
            ranked_sub_chunks = self._retrieve_with_hyde(query)
        else:
            ranked_sub_chunks = self.vector_store.hybrid_search_with_rerank(
                query, k=conf.RETRIEVAL_K, source_filter=source_filter
            )
        
        return ranked_sub_chunks[:conf.CANDIDATE_M]
    
    def generate_answer(self, query, source_filter=None, history=None):
        """生成答案，支持对话历史，支持流式输出"""
        # 查询分类
        query_category = self.query_classifier.predict_category(query)
        
        if query_category == "通用知识":
            context = ""
        else:
            # RAG检索
            strategy = self.strategy_selector.select_strategy(query)
            context_docs = self.retrieve_and_merge(query, source_filter=source_filter, strategy=strategy)
            context = "\n\n".join([doc.page_content for doc in context_docs]) if context_docs else ""
        
        # 构造Prompt
        prompt_input = self.rag_prompt.format(
            context=context,
            history=history_context,
            question=query,
            phone=conf.CUSTOMER_SERVICE_PHONE
        )
        
        # 流式输出
        for token in self.llm(prompt_input):
            yield token
```

**关键设计**：
1. **查询分类**：使用BERT分类器判断是通用知识还是专业咨询
2. **多种检索策略**：HyDE、子查询、回溯问题、直接检索
3. **策略选择**：使用LLM根据查询自动选择策略

### 1.5 向量存储 (vector_store.py)

**核心代码**：
```python
from pymilvus import MilvusClient, DataType, AnnSearchRequest, WeightedRanker
from milvus_model.hybrid import BGEM3EmbeddingFunction

class VectorStore:
    def __init__(self, collection_name, host, port, database):
        # 初始化BGE-M3嵌入函数
        self.embedding_function = BGEM3EmbeddingFunction(model_name_or_path=m3_path)
        
        # 初始化Reranker
        self.reranker = CrossEncoder(reranker_path)
        
        # 初始化Milvus客户端
        self.client = MilvusClient(uri=f"http://{host}:{port}", db_name=database)
        
        self._create_or_load_collection()
    
    def _create_or_load_collection(self):
        """创建或加载Milvus集合"""
        if not self.client.has_collection(self.collection_name):
            # 创建Schema
            schema = self.client.create_schema(auto_id=False, enable_dynamic_field=True)
            schema.add_field(field_name="id", datatype=DataType.VARCHAR, is_primary=True, max_length=100)
            schema.add_field(field_name="text", datatype=DataType.VARCHAR, max_length=65535)
            schema.add_field(field_name="dense_vector", datatype=DataType.FLOAT_VECTOR, dim=self.dense_dim)
            schema.add_field(field_name="sparse_vector", datatype=DataType.SPARSE_FLOAT_VECTOR)
            schema.add_field(field_name="parent_id", datatype=DataType.VARCHAR, max_length=100)
            schema.add_field(field_name="parent_content", datatype=DataType.VARCHAR, max_length=65535)
            schema.add_field(field_name="source", datatype=DataType.VARCHAR, max_length=50)
            
            # 创建索引
            index_params = self.client.prepare_index_params()
            index_params.add_index(field_name="dense_vector", index_type="IVF_FLAT", metric_type="IP")
            index_params.add_index(field_name="sparse_vector", index_type="SPARSE_INVERTED_INDEX", metric_type="IP")
            
            self.client.create_collection(collection_name=self.collection_name, schema=schema, index_params=index_params)
        
        self.client.load_collection(self.collection_name)
    
    def hybrid_search_with_rerank(self, query, k=5, source_filter=None):
        """混合检索并重排序"""
        # 生成查询嵌入
        query_embeddings = self.embedding_function([query])
        
        # 创建稠密向量搜索请求
        dense_request = AnnSearchRequest(
            data=[query_embeddings["dense"][0]],
            anns_field="dense_vector",
            param={"metric_type": "IP", "params": {"nprobe": 10}},
            limit=k
        )
        
        # 创建稀疏向量搜索请求
        sparse_request = AnnSearchRequest(
            data=[query_embeddings["sparse"][0]],
            anns_field="sparse_vector",
            param={"metric_type": "IP"},
            limit=k
        )
        
        # 加权排序：稀疏向量权重0.7，稠密向量权重0.3
        ranker = WeightedRanker(0.7, 0.3)
        
        # 执行混合搜索
        results = self.client.hybrid_search(
            collection_name=self.collection_name,
            reqs=[dense_request, sparse_request],
            ranker=ranker,
            limit=k
        )[0]
        
        # 转换为Document对象
        sub_chunks = [self._doc_from_hit(hit["entity"]) for hit in results]
        
        # 提取去重的父文档
        parent_docs = self._get_unique_parent_docs(sub_chunks)
        
        # 重排序
        if len(parent_docs) >= 2:
            pairs = [[query, doc.page_content] for doc in parent_docs]
            scores = self.reranker.predict(pairs)
            ranked_parent_docs = [doc for _, doc in sorted(zip(scores, parent_docs), reverse=True)]
        else:
            ranked_parent_docs = parent_docs
        
        return ranked_parent_docs[:conf.CANDIDATE_M]
```

**关键设计**：
1. **BGE-M3嵌入**：支持稠密向量和稀疏向量
2. **混合检索**：结合稠密向量（语义）和稀疏向量（关键词）
3. **重排序**：使用BGE-Reranker进行重排序
4. **父子块**：存储子块和父块，检索子块但返回父块

### 1.6 查询分类器 (query_classifier.py)

**核心代码**：
```python
from transformers import BertTokenizer, BertForSequenceClassification

class QueryClassifier:
    def __init__(self, model_path="bert_query_classifier"):
        self.tokenizer = BertTokenizer.from_pretrained(model_path)
        self.model = BertForSequenceClassification.from_pretrained(model_path, num_labels=2)
        self.label_map = {"通用知识": 0, "专业咨询": 1}
    
    def predict_category(self, query):
        """预测查询类别"""
        encoding = self.tokenizer(query, truncation=True, padding=True, max_length=128, return_tensors="pt")
        encoding = {k: v.to(self.device) for k, v in encoding.items()}
        
        self.model.eval()
        with torch.no_grad():
            outputs = self.model(**encoding)
            prediction = torch.argmax(outputs.logits, dim=1).item()
        
        return "专业咨询" if prediction == 1 else "通用知识"
```

**关键设计**：
1. **BERT分类**：使用预训练的BERT模型进行分类
2. **二分类**：通用知识 vs 专业咨询
3. **GPU加速**：自动检测并使用GPU

### 1.7 策略选择器 (strategy_selector.py)

**核心代码**：
```python
class StrategySelector:
    def __init__(self):
        self.client = OpenAI(api_key=..., base_url=...)
        self.strategy_prompt_template = self._get_strategy_prompt()
    
    def select_strategy(self, query):
        """选择检索策略"""
        strategy = self.call_dashscope(self.strategy_prompt_template.format(query=query)).strip()
        return strategy
    
    def _get_strategy_prompt(self):
        return PromptTemplate(
            template="""
            你是一个智能助手，负责分析用户查询 {query}，并从以下四种检索增强策略中选择一个最适合的策略：
            
            1. 直接检索：查询意图明确，需要从知识库中检索特定信息
            2. 假设问题检索（HyDE）：查询较为抽象，直接检索效果不佳
            3. 子查询检索：查询涉及多个实体或方面
            4. 回溯问题检索：查询较为复杂，需要简化后才能有效检索
            
            根据用户查询 {query}，直接返回最适合的策略名称。
            """,
            input_variables=["query"],
        )
```

**关键设计**：
1. **LLM选择策略**：使用LLM分析查询并选择策略
2. **四种策略**：直接检索、HyDE、子查询、回溯问题
3. **自动决策**：根据查询复杂度自动选择

## 三、关键设计模式

### 3.1 双系统架构
```
用户查询
    ↓
┌─────────────────────────────┐
│      集成系统 (main.py)     │
└─────────────────────────────┘
    ↓           ↓
┌─────────┐ ┌─────────────────┐
│ MySQL   │ │ Milvus RAG      │
│ FAQ系统 │ │ 系统            │
│ (BM25)  │ │ (向量检索+LLM) │
└─────────┘ └─────────────────┘
```

**优势**：
- 简单问题走BM25，快速返回
- 复杂问题走RAG，语义理解
- 缓解RAG系统压力

查询 → Redis缓存 → 命中则返回

                → 结果缓存到Redis
