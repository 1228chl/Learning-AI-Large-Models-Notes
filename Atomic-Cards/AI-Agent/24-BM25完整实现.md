## 基础 BM25

```python
from rank_bm25 import BM25Okapi
import jieba
import numpy as np

# 1. 准备文档
documents = [
    "北京大学创办于1898年",
    "清华大学是中国著名高等学府",
    "人工智能是计算机科学的分支",
]
tokenized_docs = [list(jieba.cut(doc)) for doc in documents]

# 2. 创建 BM25 索引
bm25 = BM25Okapi(tokenized_docs)

# 3. 检索
query = "北京大学什么时候成立的"
query_tokens = list(jieba.cut(query))
scores = bm25.get_scores(query_tokens)     # 每个文档的得分

# 4. Softmax 归一化（分数映射到 0-1 范围）
def softmax(scores):
    exp_scores = np.exp(scores - np.max(scores))
    return exp_scores / exp_scores.sum()

normalized_scores = softmax(scores)

# 5. 阈值过滤
threshold = 0.85
best_idx = np.argmax(normalized_scores)
if normalized_scores[best_idx] > threshold:
    print(f"匹配到: {documents[best_idx]}")
else:
    print("未找到高匹配度的答案")
```

## 生产级 BM25 + Redis 缓存

```python
import hashlib
import json

class BM25Search:
    """带 Redis 缓存的 BM25 搜索引擎"""

    def __init__(self, redis_client, mysql_client):
        self.redis_client = redis_client
        self.mysql_client = mysql_client
        self.bm25 = None
        self.questions = []
        self.original_questions = []
        self._load_data()

    def _load_data(self):
        """从 Redis（缓存）或 MySQL 加载问题数据"""
        # 先从 Redis 获取缓存
        original_key = "qa_original_questions"
        tokenized_key = "qa_tokenized_questions"

        self.original_questions = self.redis_client.get_data(original_key)
        tokenized_questions = self.redis_client.get_data(tokenized_key)

        if not self.original_questions or not tokenized_questions:
            # Redis 无缓存 → 从 MySQL 加载
            self.original_questions = self.mysql_client.fetch_questions()
            tokenized_questions = [
                self._tokenize(q[0]) for q in self.original_questions
            ]
            # 写入 Redis 缓存
            self.redis_client.set_data(original_key, self.original_questions)
            self.redis_client.set_data(tokenized_key, tokenized_questions)

        self.questions = tokenized_questions
        self.bm25 = BM25Okapi(self.questions)

    def _tokenize(self, text):
        """中文分词"""
        import jieba
        return list(jieba.cut(text))

    def _softmax(self, scores):
        """归一化 BM25 分数到 0~1"""
        exp_scores = np.exp(scores - np.max(scores))
        return exp_scores / exp_scores.sum()

    def search(self, query, threshold=0.85):
        """检索并返回答案"""
        if not query or not isinstance(query, str):
            return None, False

        # 检查查询缓存（是否已有答案）
        cached = self.redis_client.get_answer(query)
        if cached:
            return cached, False

        # BM25 检索
        query_tokens = self._tokenize(query)
        scores = self.bm25.get_scores(query_tokens)
        normalized = self._softmax(scores)

        best_idx = np.argmax(normalized)
        best_score = normalized[best_idx]

        if best_score > threshold:
            return best_idx, True
        return None, False
```

## 文本预处理（中文）

```python
import jieba
import re

def preprocess_text(text):
    """中文文本预处理：分词 + 去停用词"""
    # 去除非中文字符和数字
    text = re.sub(r'[^一-龥a-zA-Z]', ' ', text)
    # 分词
    words = jieba.lcut(text)
    # 过滤单字和停用词（可根据需要扩展）
    stop_words = {'的', '了', '在', '是', '我', '有', '和', '就', '不', '人', '都', '一', '一个', '上', '也', '很', '到', '说', '要', '去', '你', '会', '着', '没有', '看', '好', '自己', '这'}
    return [w for w in words if len(w) > 1 and w not in stop_words]
```
